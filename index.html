Google Apps Script Integration (Code.gs)

Copy and paste this code into your Google Sheet's Apps Script editor to receive submissions in real time: 

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Create Headers on first run
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Timestamp", "Course", "Registration Type", "Group Name", 
        "Total Members", "Member Details", "Topic / Title", "Supervisor"
      ]);
      sheet.getRange(1, 1, 1, 8).setFontWeight("bold");
    }
    
    // Format Members List
    var memberDetailsStr = "";
    if (data.members && data.members.length > 0) {
      memberDetailsStr = data.members.map(function(m, idx) {
        return "M" + (idx+1) + ": " + m.name + " (" + m.roll + ")";
      }).join("; ");
    }
    
    sheet.appendRow([
      data.timestamp || new Date().toLocaleString(),
      data.course || "N/A",
      data.regType || "Group Registration",
      data.groupName || "N/A",
      data.memberCount || (data.members ? data.members.length : 0),
      memberDetailsStr,
      data.topic || "N/A",
      data.supervisor || "N/A"
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({"result": "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "message": err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
  1. Open Google Sheet > Extensions > Apps Script.
  2. Paste the code into Code.gs and click Save.
  3. Deploy > New deployment > Select Web app.
  4. Set Execute as: Me and Who has access: Anyone.
  5. Deploy and paste the Web App URL into the Faculty Dashboard input field!
